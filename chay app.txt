# Cài đặt môi trường Python 3.11 (tương thích với OpenCV)

# Bước 1: Thiết lập cache directories
$env:UV_CACHE_DIR="E:\uv_cache"
$env:PIP_CACHE_DIR="E:\pip_cache"

# Deactivate venv hiện tại
deactivate

# Xóa venv
Remove-Item -Path .venv -Recurse -Force

# Tạo lại với Python 3.11
uv venv --python 3.11

# Kích hoạt
.venv\Scripts\activate

# Cài packages
pip install --upgrade pip setuptools wheel
pip uninstall opencv-python opencv-contrib-python opencv-python-headless -y
pip install numpy==1.24.3
pip install opencv-python==4.8.1.78 --no-cache-dir

# Install PyTorch (CPU version for compatibility)
pip install torch torchvision torchaudio

# Install other dependencies
pip install albumentations

# Kiểm tra cài đặt
python --version
python -c "import sys; print(f'Python executable: {sys.executable}')"
python -c "import numpy; print(f'NumPy: {numpy.__version__}')"
python -c "import cv2; print(f'OpenCV: {cv2.__version__}')"

# Nếu cv2 vẫn không import được, thử:
# pip list | findstr opencv
# pip show opencv-python

# Chạy script
python src/utils/collect_images.py






# Labelling them 

pip install label-studio


# Training 🦾
1. Create a checkpoints folder `mkdir checkpoints`
2. Run the training pipeline `uv run src/train.py`

# Running  🚀 
1. To test on your test set, update the checkpoint parameter in `test.py` then run `uv run src/test.py`
2. To run in real time, update the checkpoint parameter in `realtime.py` then run `uv run src/realtime.py
